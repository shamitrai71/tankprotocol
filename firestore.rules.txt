rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }

    function myRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }

    function isAdmin() {
      return isSignedIn() && (myRole() == 'admin' || myRole() == 'super_admin');
    }

    function isSuperAdmin() {
      return isSignedIn() && myRole() == 'super_admin';
    }

    // Current quota period as "YYYY-MM", derived from SERVER time so a client
    // cannot claim a fabricated month to win itself a fresh allowance.
    function currentPeriod() {
      return string(request.time.year()) + '-' +
        (request.time.month() < 10
          ? '0' + string(request.time.month())
          : string(request.time.month()));
    }

    // The calculator increments the member's own counter after a download, so
    // the field cannot simply be read-only. Instead it may move in exactly
    // three legitimate ways - never downward, and never into a period the
    // server clock disagrees with:
    //   1. unchanged
    //   2. +1 within the SAME period (a normal download)
    //   3. reset to 1 when a genuinely new period has begun (first download
    //      of the new month), validated against server time
    // This blocks the obvious attacks: zeroing your own counter, or inventing
    // a future month to earn a fresh allowance.
    function quotaChangeOk() {
      return (request.resource.data.get('downloadsUsed', 0) == resource.data.get('downloadsUsed', 0)
              && request.resource.data.get('downloadsPeriod', '') == resource.data.get('downloadsPeriod', ''))
          || (request.resource.data.get('downloadsUsed', 0) == resource.data.get('downloadsUsed', 0) + 1
              && request.resource.data.get('downloadsPeriod', '') == resource.data.get('downloadsPeriod', ''))
          || (request.resource.data.get('downloadsUsed', 0) == 1
              && request.resource.data.get('downloadsPeriod', '') == currentPeriod()
              && resource.data.get('downloadsPeriod', '') != currentPeriod());
    }

    match /users/{uid} {

      // Own doc always readable; admins and super admins may read all.
      allow read: if isSignedIn() && (request.auth.uid == uid || isAdmin());

      // Self-signup: cannot self-promote, self-approve, or start with credit
      // already granted.
      allow create: if isSignedIn()
                    && request.auth.uid == uid
                    && request.resource.data.role == 'user'
                    && request.resource.data.approved == false
                    && request.resource.data.get('downloadsUsed', 0) == 0;

      // Self-service profile updates. A member may edit their own profile and
      // record acceptance of the Terms, but may NOT touch anything governing
      // their own privileges:
      //   role               - privilege escalation
      //   approved           - self-approval
      //   canGenerateReports - would bypass a super admin's report block
      //   downloadLimit      - quota override
      //   downloadsUsed / downloadsPeriod - constrained by quotaChangeOk()
      allow update: if isSignedIn()
                    && request.auth.uid == uid
                    && request.resource.data.role == resource.data.role
                    && request.resource.data.approved == resource.data.approved
                    && request.resource.data.get('canGenerateReports', true)
                       == resource.data.get('canGenerateReports', true)
                    && request.resource.data.get('downloadLimit', 0)
                       == resource.data.get('downloadLimit', 0)
                    && quotaChangeOk();

      // Only a super admin may change roles, approval, permissions, or delete.
      allow update, delete: if isSuperAdmin();
    }

    match /reports/{id} {

      // A report is "research" (private practice/testing, the default - also
      // the fallback for any report written before this field existed) or
      // "official" (a true record, visible to the rest of the company).
      // Admins can always read everything either way.
      function sameCompanyReport() {
        return isSignedIn() &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.company == resource.data.company;
      }

      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if isSignedIn() && (
        resource.data.uid == request.auth.uid ||
        isAdmin() ||
        (resource.data.get('recordType','research') == 'official' && sameCompanyReport())
      );
      // Members may only flag their own report as downloaded - nothing else.
      allow update: if isSignedIn()
                    && resource.data.uid == request.auth.uid
                    && request.resource.data.diff(resource.data)
                         .affectedKeys().hasOnly(['downloaded']);
      allow delete: if false;
    }

      // `ventingCalcs`/`settlementCalcs`/`shellThicknessCalcs`/
      // `inspectionIntervalCalcs`/`brittleFractureCalcs`/
      // `strappingTableCalcs`/`windGirderCalcs`/`roofVentCalcs`/
      // `seismicCalcs`/`freeboardCalcs`/`shellDesignCalcs`/
      // `roofDesignCalcs`/`nozzleReinforcementCalcs`/`bottomPlateCalcs`/
      // `nozzleLoadsCalcs`: auto-logged on each calculation, "research" by
      // default (private), "official" visible to the rest of the company,
      // admins can always read either way. Only the `downloaded` flag is
      // mutable after save.
      function sameCompanyFrangible() {
        return isSignedIn() &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.company == resource.data.company;
      }

      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if isSignedIn() && (
        resource.data.uid == request.auth.uid ||
        isAdmin() ||
        (resource.data.get('recordType','research') == 'official' && sameCompanyFrangible())
      );
      allow update: if isSignedIn()
                    && resource.data.uid == request.auth.uid
                    && request.resource.data.diff(resource.data)
                         .affectedKeys().hasOnly(['downloaded']);
      allow delete: if false;
    }

    match /rimSealSurveys/{id} {

      // Colleagues at the same company need to see each other's surveys on the
      // same tank to compare gap positions across levels/dates - unlike
      // `reports`, this isn't purely personal history, so read access is
      // scoped to "same company" (via the requester's own user doc), not just
      // "own uid" - but ONLY for surveys marked "official". A "research"
      // survey (the default) stays private to its author, same idea as
      // `reports`. Admins can always read everything either way.
      function sameCompany() {
        return isSignedIn() &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.company == resource.data.company;
      }

      allow create: if isSignedIn()
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.status == 'draft';

      allow read: if isSignedIn() && (
        resource.data.uid == request.auth.uid ||
        isAdmin() ||
        (resource.data.get('recordType','research') == 'official' && sameCompany())
      );

      // A survey is an inspection record: once submitted, the measured
      // readings become an immutable audit trail, same principle as
      // `reports`/`events` - full edits are only allowed while still a draft.
      // The regulatory limits applied against those readings are a reference/
      // evaluation lens, not measured data, so they stay correctable even
      // after submission - a second, narrower allowance for a diff touching
      // only those fields (mirrors how `reports` allows updating just
      // `downloaded` regardless of who logged it).
      function limitsOnlyDiff() {
        return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'regulatoryProfile', 'primaryMaxWidth', 'primaryMaxAreaPerFt',
          'secondaryMaxWidth', 'secondaryMaxAreaPerFt',
          'rimGapExpMinMm', 'rimGapExpMaxMm', 'updatedAt'
        ]);
      }

      allow update: if isSignedIn() && resource.data.uid == request.auth.uid &&
                    (resource.data.status == 'draft' || limitsOnlyDiff());
      allow delete: if isSignedIn() && resource.data.uid == request.auth.uid && resource.data.status == 'draft';
    }

    // Site analytics: a signed-in member may append events about themselves.
    // Only admins can read the log; events are immutable once written.
    match /events/{id} {
      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if isAdmin();
      allow update, delete: if false;
    }

    // Controlled vocabularies and platform settings: readable by any signed-in
    // member (the calculator and profile need them), writable only by a super
    // admin - this is what makes the report limits tamper-proof.
    match /config/{doc} {
      allow read: if isSignedIn();
      allow write: if isSuperAdmin();
    }
  }
